<script>
(function () {
  'use strict';

  /* ------------------------------------------------------------------ */
  /*  Preset colors                                                      */
  /* ------------------------------------------------------------------ */
  var PRESETS = [
    ['#000000', 'black'],
    ['#FFFFFF', 'white'],
    ['#7A7A7A', 'gray'],
    ['#E5484D', 'red'],
    ['#F76B15', 'orange'],
    ['#F5D90A', 'yellow'],
    ['#30A46C', 'green'],
    ['#12A594', 'teal'],
    ['#3E63DD', 'blue'],
    ['#8E4EC6', 'purple'],
    ['#D6409F', 'pink']
  ];

  /* ------------------------------------------------------------------ */
  /*  DOM refs                                                           */
  /* ------------------------------------------------------------------ */
  var $ = function (id) { return document.getElementById(id); };

  var inputEl        = $('input');
  var outputEl       = $('output');
  var statusEl       = $('status');
  var swatchEl       = $('swatches');
  var pickerEl       = $('picker');
  var hexEl          = $('hex');
  var minifyEl       = $('minify');
  var genBtn         = $('generate');
  var copyBtn        = $('copy');
  var dlBtn          = $('download');

  var previewEl      = $('preview');
  var previewEmptyEl = $('preview-empty');
  var lottieStage    = $('lottie-stage');
  var bgButtons      = document.querySelectorAll('.bg-btn');
  var lottieAnim     = null;

  var currentHex = '#FFFFFF';

  /* ------------------------------------------------------------------ */
  /*  Color helpers                                                      */
  /* ------------------------------------------------------------------ */
  function round4(n) { return Math.round(n * 10000) / 10000; }

  function normalizeHex(str) {
    if (typeof str !== 'string') return null;
    var s = str.trim().replace(/^#/, '');
    if (s.length === 3) s = s[0] + s[0] + s[1] + s[1] + s[2] + s[2];
    if (!/^[0-9a-fA-F]{6}$/.test(s)) return null;
    return '#' + s.toUpperCase();
  }

  function hexToRgb01(hex) {
    var h = hex.slice(1);
    return [
      round4(parseInt(h.slice(0, 2), 16) / 255),
      round4(parseInt(h.slice(2, 4), 16) / 255),
      round4(parseInt(h.slice(4, 6), 16) / 255)
    ];
  }

  function isColorArray(a) {
    return Array.isArray(a) &&
           (a.length === 3 || a.length === 4) &&
           a.every(function (v) { return typeof v === 'number' && isFinite(v); });
  }

  function makeColor(orig, rgb) {
    return orig.length === 4 ? [rgb[0], rgb[1], rgb[2], orig[3]]
                             : [rgb[0], rgb[1], rgb[2]];
  }

  /* {a:0, k:[r,g,b,a]}  or  animated keyframes */
  function applyColorHolder(holder, rgb) {
    if (!holder || typeof holder !== 'object' || Array.isArray(holder)) return false;
    var k = holder.k;
    if (!Array.isArray(k)) return false;

    if (isColorArray(k)) {
      holder.k = makeColor(k, rgb);
      return true;
    }

    var touched = false;
    for (var i = 0; i < k.length; i++) {
      var frame = k[i];
      if (!frame || typeof frame !== 'object' || Array.isArray(frame)) continue;
      if (isColorArray(frame.s)) { frame.s = makeColor(frame.s, rgb); touched = true; }
      if (isColorArray(frame.e)) { frame.e = makeColor(frame.e, rgb); touched = true; }
    }
    return touched;
  }

  /* gradient: g = { p:<stops>, k:{ a:0, k:[pos,r,g,b, pos,r,g,b, ...] } } */
  function applyGradient(g, rgb) {
    if (!g || typeof g !== 'object') return false;
    var p = g.p;
    var holder = g.k;
    if (!holder || typeof holder !== 'object') return false;
    var data = holder.k;
    if (!Array.isArray(data) || !data.length) return false;

    function paint(arr) {
      if (!Array.isArray(arr) || typeof p !== 'number' || p < 1) return false;
      if (arr.length % p !== 0) return false;
      var stride = arr.length / p;
      if (stride < 4) return false;
      for (var i = 0; i < p; i++) {
        var b = i * stride;
        arr[b + 1] = rgb[0];
        arr[b + 2] = rgb[1];
        arr[b + 3] = rgb[2];
      }
      return true;
    }

    var touched = false;
    if (typeof data[0] === 'number') {
      touched = paint(data);
    } else {
      for (var i = 0; i < data.length; i++) {
        var frame = data[i];
        if (!frame || typeof frame !== 'object') continue;
        if (paint(frame.s)) touched = true;
        if (paint(frame.e)) touched = true;
      }
    }
    return touched;
  }

  /* ------------------------------------------------------------------ */
  /*  Core walker                                                        */
  /* ------------------------------------------------------------------ */
  function colorize(data, hex) {
    var rgb = hexToRgb01(hex);
    var stats = { stroke: 0, fill: 0, gradient: 0, solid: 0, text: 0 };

    (function walk(node) {
      if (Array.isArray(node)) {
        for (var i = 0; i < node.length; i++) walk(node[i]);
        return;
      }
      if (!node || typeof node !== 'object') return;

      var ty = node.ty;

      if (ty === 'st' || ty === 'fl') {
        if (applyColorHolder(node.c, rgb)) {
          if (ty === 'st') stats.stroke++; else stats.fill++;
        }
      }
      else if (ty === 'gf' || ty === 'gs') {
        if (applyGradient(node.g, rgb)) stats.gradient++;
      }
      else if (ty === 1 && typeof node.sc === 'string') {
        node.sc = hex.toLowerCase();
        stats.solid++;
      }

      if (node.t && node.t.d && Array.isArray(node.t.d.k)) {
        for (var j = 0; j < node.t.d.k.length; j++) {
          var doc = node.t.d.k[j];
          var s = doc && doc.s;
          if (!s) continue;
          if (applyColorHolder(s.fc, rgb)) stats.text++;
          if (applyColorHolder(s.sc, rgb)) stats.text++;
        }
      }

      for (var key in node) {
        if (!Object.prototype.hasOwnProperty.call(node, key)) continue;
        var child = node[key];
        if (child && typeof child === 'object') walk(child);
      }
    })(data);

    return stats;
  }

  /* ------------------------------------------------------------------ */
  /*  Preview                                                            */
  /* ------------------------------------------------------------------ */
  function setPreviewBg(mode) {
    previewEl.className = 'preview mode-' + mode;
    for (var i = 0; i < bgButtons.length; i++) {
      bgButtons[i].setAttribute('aria-pressed',
        bgButtons[i].dataset.bg === mode ? 'true' : 'false');
    }
  }

  function clearPreview(msg) {
    if (lottieAnim) {
      try { lottieAnim.destroy(); } catch (e) { /* noop */ }
      lottieAnim = null;
    }
    lottieStage.innerHTML = '';
    lottieStage.classList.remove('has-content');
    previewEmptyEl.style.display = '';
    previewEmptyEl.textContent = msg || 'Generate to see a preview';
  }

  function renderPreview(data) {
    if (lottieAnim) {
      try { lottieAnim.destroy(); } catch (e) { /* noop */ }
      lottieAnim = null;
    }
    lottieStage.innerHTML = '';

    if (!window.lottie) {
      lottieStage.classList.remove('has-content');
      previewEmptyEl.style.display = '';
      previewEmptyEl.textContent = 'lottie-web failed to load — check network';
      return;
    }

    /* clone so any lottie-internal mutation can't affect the output textarea */
    var cloned;
    try { cloned = JSON.parse(JSON.stringify(data)); }
    catch (e) {
      previewEmptyEl.style.display = '';
      previewEmptyEl.textContent = 'Render error: ' + e.message;
      return;
    }

    try {
      lottieAnim = window.lottie.loadAnimation({
        container: lottieStage,
        renderer: 'svg',
        loop: true,
        autoplay: true,
        animationData: cloned,
        rendererSettings: { preserveAspectRatio: 'xMidYMid meet' }
      });
      lottieStage.classList.add('has-content');
      previewEmptyEl.style.display = 'none';
    } catch (err) {
      lottieStage.classList.remove('has-content');
      previewEmptyEl.style.display = '';
      previewEmptyEl.textContent = 'Render error: ' + err.message;
    }
  }

  /* ------------------------------------------------------------------ */
  /*  UI                                                                 */
  /* ------------------------------------------------------------------ */
  function setStatus(msg, isErr) {
    statusEl.textContent = msg;
    statusEl.className = 'status' + (isErr ? ' err' : '');
  }

  function syncSwatches() {
    var list = swatchEl.querySelectorAll('.swatch');
    for (var i = 0; i < list.length; i++) {
      var el = list[i];
      el.setAttribute('aria-pressed',
        el.dataset.hex.toUpperCase() === currentHex ? 'true' : 'false');
    }
  }

  function setColor(hex, opts) {
    opts = opts || {};
    var n = normalizeHex(hex);
    if (!n) return false;
    currentHex = n;
    if (!opts.skipPicker)   pickerEl.value = n.toLowerCase();
    if (!opts.skipHex)      hexEl.value = n;
    if (!opts.skipSwatches) syncSwatches();
    return true;
  }

  function buildSwatches() {
    PRESETS.forEach(function (p) {
      var b = document.createElement('button');
      b.type = 'button';
      b.className = 'swatch';
      b.style.background = p[0];
      b.dataset.hex = p[0];
      b.title = p[1] + '  ' + p[0];
      b.setAttribute('aria-label', p[1]);
      b.setAttribute('aria-pressed', 'false');
      b.addEventListener('click', function () { setColor(p[0]); });
      swatchEl.appendChild(b);
    });
  }

  function generate() {
    var raw = inputEl.value.trim();
    if (!raw) {
      setStatus('No input — paste a JSON icon first', true);
      return;
    }

    var parsed;
    try {
      parsed = JSON.parse(raw);
    } catch (err) {
      outputEl.value = '';
      setStatus('Invalid JSON — ' + err.message, true);
      return;
    }

    var stats = colorize(parsed, currentHex);
    var total = stats.stroke + stats.fill + stats.gradient + stats.solid + stats.text;

    outputEl.value = minifyEl.checked
      ? JSON.stringify(parsed)
      : JSON.stringify(parsed, null, 2);

    var parts = [];
    if (stats.stroke)   parts.push(stats.stroke + '× stroke');
    if (stats.fill)     parts.push(stats.fill + '× fill');
    if (stats.gradient) parts.push(stats.gradient + '× gradient');
    if (stats.solid)    parts.push(stats.solid + '× solid');
    if (stats.text)     parts.push(stats.text + '× text');

    if (!total) {
      setStatus('⚠ No color properties found in this file', true);
    } else {
      setStatus('OK — ' + total + ' color properties → ' + currentHex +
                '  (' + parts.join(', ') + ')', false);
    }

    renderPreview(parsed);
  }

  function copyOut() {
    var text = outputEl.value;
    if (!text) { setStatus('Nothing to copy', true); return; }

    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(function () {
        setStatus('Copied to clipboard', false);
      }, function () { legacyCopy(text); });
    } else {
      legacyCopy(text);
    }
  }

  function legacyCopy(text) {
    outputEl.removeAttribute('readonly');
    outputEl.select();
    try { document.execCommand('copy'); setStatus('Copied to clipboard', false); }
    catch (e) { setStatus('Copy failed — select manually', true); }
    outputEl.setAttribute('readonly', 'readonly');
    outputEl.setSelectionRange(0, 0);
  }

  function download() {
    var text = outputEl.value;
    if (!text) { setStatus('Nothing to download', true); return; }

    var name = 'icon-colored.json';
    try {
      var obj = JSON.parse(inputEl.value);
      if (obj && typeof obj.nm === 'string' && obj.nm.trim()) {
        name = obj.nm.trim().replace(/[^\w.-]+/g, '_') + '-colored.json';
      }
    } catch (e) { /* ignore */ }

    var blob = new Blob([text], { type: 'application/json' });
    var url = URL.createObjectURL(blob);
    var a = document.createElement('a');
    a.href = url;
    a.download = name;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(function () { URL.revokeObjectURL(url); }, 1000);
    setStatus('Downloaded ' + name, false);
  }

  /* ------------------------------------------------------------------ */
  /*  Wiring                                                             */
  /* ------------------------------------------------------------------ */
  buildSwatches();
  setColor('#FFFFFF');
  setStatus('Ready.');
  clearPreview('Generate to see a preview');

  for (var i = 0; i < bgButtons.length; i++) {
    (function (btn) {
      btn.addEventListener('click', function () {
        setPreviewBg(btn.dataset.bg);
      });
    })(bgButtons[i]);
  }

  genBtn.addEventListener('click', generate);
  copyBtn.addEventListener('click', copyOut);
  dlBtn.addEventListener('click', download);

  pickerEl.addEventListener('input', function () {
    setColor(pickerEl.value);
  });

  hexEl.addEventListener('input', function () {
    setColor(hexEl.value, { skipHex: true });
  });
  hexEl.addEventListener('blur', function () {
    hexEl.value = currentHex;
  });

  document.addEventListener('keydown', function (e) {
    if ((e.metaKey || e.ctrlKey) && e.key === 'Enter') {
      e.preventDefault();
      generate();
    }
  });
})();
</script>
