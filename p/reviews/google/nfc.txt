        document.addEventListener('DOMContentLoaded', () => {
            const track1 = document.getElementById('track1');
            const track2 = document.getElementById('track2');
            
            // Split data into two columns for the tracks
            const col1Data = reviewsData.filter((_, i) => i % 2 === 0);
            const col2Data = reviewsData.filter((_, i) => i % 2 !== 0);

            // Function to populate a track multiple times to create enough content for seamless scrolling
            function populateTrack(track, data) {
                let html = '';
                // Duplicate data to ensure track is taller than container
                for(let i=0; i<3; i++) {
                    data.forEach(review => {
                        html += createReviewCard(review);
                    });
                }
                track.innerHTML = html;
            }

            populateTrack(track1, col1Data);
            populateTrack(track2, col2Data);

            // Setup scrolling animation variables
            let scrollPos1 = 0;
            // Start track 2 halfway to offset the cards visually
            let scrollPos2 = -150; 
            const speed = 0.5; // Pixels per frame

            function animateScroll() {
                scrollPos1 -= speed;
                // Reverse direction for track 2
                scrollPos2 += speed;

                const resetPoint = -(track1.scrollHeight / 3);
                // For track 2 going up, reset when it goes above 0
                const resetPointUp = 0; 
                // We need to know max scroll for upward scrolling reset
                const maxScrollUp = -(track2.scrollHeight / 3);

                if (scrollPos1 <= resetPoint) {
                    scrollPos1 = 0;
                }
                
                if (scrollPos2 >= resetPointUp) {
                    // Reset to the negative position representing the bottom of the first block
                    scrollPos2 = maxScrollUp;
                }

                track1.style.transform = `translateY(${scrollPos1}px)`;
                track2.style.transform = `translateY(${scrollPos2}px)`;

                requestAnimationFrame(animateScroll);
            }

            // Start animation
            requestAnimationFrame(animateScroll);
        });

        function openReviewModal() {
            // Set iframe src just before opening to avoid loading it on initial page load
            if (!iframe.src || iframe.src === window.location.href) {
                iframe.src = reviewUrl;
            }

            // Show overlay
            modalOverlay.classList.remove('hidden');
            
            // Small delay to allow display:block to apply before animating opacity
            setTimeout(() => {
                modalOverlay.classList.remove('opacity-0');
                modalOverlay.classList.add('opacity-100');
                
                modal.classList.remove('scale-95');
                modal.classList.add('scale-100');
            }, 10);
            
            // Prevent body scrolling
            document.body.style.overflow = 'hidden';
        }

        function closeReviewModal() {
            // Animate out
            modalOverlay.classList.remove('opacity-100');
            modalOverlay.classList.add('opacity-0');
            
            modal.classList.remove('scale-100');
            modal.classList.add('scale-95');

            // Hide after animation completes
            setTimeout(() => {
                modalOverlay.classList.add('hidden');
                // Optional: clear iframe to stop any background processes if needed
                // iframe.src = ''; 
            }, 300); // Matches duration-300 class

            // Restore body scrolling
            document.body.style.overflow = 'auto';
        }

        // Close on click outside modal content
        modalOverlay.addEventListener('click', (e) => {
            if (e.target === modalOverlay) {
                closeReviewModal();
            }
        });
        
        // Close on escape key
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape' && !modalOverlay.classList.contains('hidden')) {
                closeReviewModal();
            }
        });
