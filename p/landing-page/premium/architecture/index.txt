    (function initHeroSlider() {
      const sliderEl = document.getElementById('heroSlider');
      const SLIDE_INTERVAL = 6500; // ms per slide
      const FADE_DURATION = 1800;  // matches CSS opacity transition

      const slides = DEMO_IMAGES.heroSlides.map((item, index) => {
        const div = document.createElement('div');
        div.className = 'hero__slide' + (index === 0 ? ' is-active' : '');
        const img = document.createElement('img');
        img.src = item.src;
        img.alt = item.alt;
        div.appendChild(img);
        sliderEl.appendChild(div);
        return div;
      });

      // Trigger smooth zoom on initial slide immediately upon load
      requestAnimationFrame(() => {
        requestAnimationFrame(() => {
          if (slides[0]) slides[0].classList.add('is-zooming');
        });
      });

      if (slides.length < 2) return;

      let current = 0;
      setInterval(() => {
        const prevIndex = current;
        current = (current + 1) % slides.length;

        const prevSlide = slides[prevIndex];
        const nextSlide = slides[current];

        // Start crossfade + slow zoom on next slide
        nextSlide.classList.add('is-active');
        requestAnimationFrame(() => {
          nextSlide.classList.add('is-zooming');
        });

        // Fade out previous slide while keeping its zoom until fully invisible
        prevSlide.classList.remove('is-active');
        setTimeout(() => {
          prevSlide.classList.remove('is-zooming');
        }, FADE_DURATION + 100);
      }, SLIDE_INTERVAL);
    })();
