<script>
  // Grab every pinned layer (all but the last) with its inner + shade + next sibling
  const layers = [...document.querySelectorAll('.stack > .layer')];
  const pinned = layers.slice(0, -1).map((layer, i) => ({
    inner: layer.querySelector('.layer__inner'),
    shade: layer.querySelector('.layer__shade'),
    next:  layers[i + 1],
  }));

  const RECEDE_SCALE = 0.9;   // how small the covered card gets
  const RECEDE_SHADE = 0.55;  // how dark the covered card gets

  // On phones the scale is jarring (the card is full-width), so shade only.
  const phone = window.matchMedia('(max-width: 639px)');

  let raf = null;
  function apply() {
    const vh = window.innerHeight || 1;
    const shrink = phone.matches ? 0 : 1 - RECEDE_SCALE;

    for (const { inner, shade, next } of pinned) {
      // p = 0  → the next layer is still a full viewport away (nothing covered yet)
      // p = 1  → the next layer has taken the whole viewport
      const p = Math.min(1, Math.max(0, 1 - next.getBoundingClientRect().top / vh));

      // No transform at rest. A persistent scale(1) would make this layer a
      // containing block for any position:fixed children (like a loader),
      // pinning them to the card instead of the viewport.
      inner.style.transform = p > 0 && shrink > 0 ? `scale(${1 - shrink * p})` : '';
      inner.style.willChange = p > 0 && shrink > 0 ? 'transform' : '';

      shade.style.opacity = String(RECEDE_SHADE * p);

      // Fully covered: stop painting it (saves work for heavy content)
      inner.style.visibility = p >= 1 ? 'hidden' : 'visible';
    }
  }

  // Coalesce scroll into one rAF
  window.addEventListener('scroll', () => {
    if (raf) cancelAnimationFrame(raf);
    raf = requestAnimationFrame(apply);
  }, { passive: true });
  window.addEventListener('resize', apply);

  apply();
</script>
